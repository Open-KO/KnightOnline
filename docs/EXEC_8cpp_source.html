19bool EXEC::Parse(const char* line, const std::string& filename, int lineNumber)
20{
21 int index = 0, argsToParse = 0;
22 bool handled = true;
23 char temp[1024] {};
24
25 ParseSpace(temp, line, index);
26
27 size_t opcode = hashing::djb2::hash(std::string_view(temp));
28 switch (opcode)
29 {
30 // E SAY {'up' event ID} {'ok' event ID} {talk ID 1} {talk ID 2} {talk ID 3} {talk ID 4} {talk ID 5} {talk ID 6} {talk ID 7} {talk ID 8}
31 // The talk ID refers to the ID in the Quest_Talk TBL in the client.
32 case "SAY"_djb2:
33 m_Exec = EXEC_SAY;
34 argsToParse = 10;
35 break;
36
37 // E SELECT_MSG {npc prototype ID - unused} {talk ID} ...
38 // ... {menu ID 1} {event ID 1} {menu ID 2} {event ID 2} {menu ID 3} {event ID 3} {menu ID 4} {event ID 4} {menu ID 5} {event ID 5}
39 // ... {menu ID 6} {event ID 6} {menu ID 7} {event ID 7} {menu ID 8} {event ID 8} {menu ID 9} {event ID 9} {menu ID 10} {event ID 10}
40 // The talk ID refers to the ID in the Quest_Talk TBL in the client. This is used for the dialogue text.
41 // The menu ID refers to the ID in the Quest_Menu TBL in the client. This is the button text.
42 // The event ID refers to the associated event ID to run on the server when this button is pressed in the client.
43 case "SELECT_MSG"_djb2:
44 m_Exec = EXEC_SELECT_MSG;
45 argsToParse = 22;
46 break;
47
48 // E RUN_EVENT {event ID}
49 case "RUN_EVENT"_djb2:
50 m_Exec = EXEC_RUN_EVENT;
51 argsToParse = 1;
52 break;
53
54 // E GIVE_ITEM {item ID} {item count}
55 case "GIVE_ITEM"_djb2:
56 m_Exec = EXEC_GIVE_ITEM;
57 argsToParse = 2;
58 break;
59
60 // E ROB_ITEM {item ID} {item count}
61 case "ROB_ITEM"_djb2:
62 m_Exec = EXEC_ROB_ITEM;
63 argsToParse = 2;
64 break;
65
66 // E OPEN_EDITBOX {npc prototype ID - unused} {input message} {next event}
67 case "OPEN_EDITBOX"_djb2:
68 m_Exec = EXEC_OPEN_EDITBOX;
69 argsToParse = 3;
70 break;
71
72 // E GIVE_NOAH {amount}
73 case "GIVE_NOAH"_djb2:
74 m_Exec = EXEC_GIVE_NOAH;
75 argsToParse = 1;
76 break;
77
78 // E LOG_COUPON_ITEM {item ID} {item count}
79 case "LOG_COUPON_ITEM"_djb2:
80 m_Exec = EXEC_LOG_COUPON_ITEM;
81 argsToParse = 2;
82 break;
83
84 // E ZONE_CHANGE {zone ID} {x} {z}
85 case "ZONE_CHANGE"_djb2:
86 m_Exec = EXEC_ZONE_CHANGE;
87 argsToParse = 3;
88 break;
89
90 // E RETURN
91 case "RETURN"_djb2:
92 m_Exec = EXEC_RETURN;
93 break;
94
95 // E PROMOTE_USER_NOVICE
96 case "PROMOTE_USER_NOVICE"_djb2:
97 m_Exec = EXEC_PROMOTE_USER_NOVICE;
98 break;
99
100 // E PROMOTE_USER
101 case "PROMOTE_USER"_djb2:
102 m_Exec = EXEC_PROMOTE_USER;
103 break;
104
105 // E SKILL_POINT_DISTRIBUTE
106 case "SKILL_POINT_DISTRIBUTE"_djb2:
107 m_Exec = EXEC_SKILL_POINT_DISTRIBUTE;
108 break;
109
110 // E SKILL_POINT_DISTRIBUTE
111 case "STAT_POINT_DISTRIBUTE"_djb2:
112 m_Exec = EXEC_STAT_POINT_DISTRIBUTE;
113 break;
114
115 // E SKILL_POINT_FREE
116 case "SKILL_POINT_FREE"_djb2:
117 m_Exec = EXEC_SKILL_POINT_FREE;
118 break;
119
120 // E STAT_POINT_FREE
121 case "STAT_POINT_FREE"_djb2:
122 m_Exec = EXEC_STAT_POINT_FREE;
123 break;
124
125 default:
126 spdlog::warn("EXEC::Parse: unhandled opcode '{}' ({}:{})", temp, filename, lineNumber);
127 handled = false;
128 break;
129 }
130
131 assert(argsToParse >= 0 && argsToParse <= MAX_EXEC_INT);
132 for (int i = 0; i < argsToParse; i++)
133 {
134 ParseSpace(temp, line, index);
135 m_ExecInt[i] = atoi(temp);
136 }
137
138 return handled;
139}
