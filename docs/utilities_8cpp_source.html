160bool ParseSpace(char* tBuf, const char* sBuf, int& bufferIndex)
161{
162 int i = 0, index = 0;
163 bool flag = false;
164
165 sBuf += bufferIndex;
166
167 while (sBuf[index] == ' ' || sBuf[index] == '\t')
168 index++;
169
170 while (sBuf[index] != ' ' && sBuf[index] != '\t' && sBuf[index] != (uint8_t) 0)
171 {
172 tBuf[i++] = sBuf[index++];
173 flag = true;
174 }
175 tBuf[i] = 0;
176
177 while (sBuf[index] == ' ' || sBuf[index] == '\t')
178 index++;
179
180 if (!flag)
181 return false;
182
183 bufferIndex += index;
184 return true;
185}

199int myrand_generic_impl(int min, int max)
200{
201 if (min == max)
202 return min;
203
204 if (min > max)
205 std::swap(min, max);
206
207 double gap = max - min + 1;
208 double rrr = static_cast<double>(RAND_MAX) / gap;
209
210 int rand_result = static_cast<int>(static_cast<double>(rand()) / rrr);
211 if (min > (INT_MAX - rand_result))
212 return max;
213
214 return std::clamp(min + rand_result, min, max);
215}
