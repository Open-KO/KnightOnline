62// This test covers CUser::CheckExchange. CheckExchange checks to make sure
63// that an ITEM_EXCHANGE record exists, and that the user has enough inventory
64// space to be granted any rewards.
65TEST_F(ItemExchangeTest, CheckExchange)
66{
67 _user->ClearInventory();
68 _user->ResetSend();
69
70 // Should return false on a db miss
71 EXPECT_FALSE(_user->CheckExchange(EXCHANGE_MISSING));
72
73 // For EXCHANGE_TYPE_ONE_OF_WEIGHTED and EXCHANGE_TYPE_ONE_OF_EQUAL
74 // There must be at least one open inventory slot to accept the result
75 // of the exchange. User inventory should currently be empty, so
76 // the checks should pass
77 EXPECT_TRUE(_user->CheckExchange(EXCHANGE_WARRIOR_BEGINNER));
78 EXPECT_TRUE(_user->CheckExchange(EXCHANGE_ISAAC_KISS));
79 EXPECT_TRUE(_user->CheckExchange(EXCHANGE_ABYSS_GEM));
80
81 // We'll fill the user's inventory with non-stacking items.
82 int emptySlots = _user->GetNumberOfEmptySlots();
83 for (int i = 0; i < emptySlots; i++)
84 {
85 _user->TestGiveItem(ITEM_GOLD_BAR, 1);
86 }
87 ASSERT_EQ(_user->GetNumberOfEmptySlots(), 0);
88
89 // Should return false with full inventory
90 EXPECT_FALSE(_user->CheckExchange(EXCHANGE_WARRIOR_BEGINNER));
91 EXPECT_FALSE(_user->CheckExchange(EXCHANGE_ISAAC_KISS));
92 EXPECT_FALSE(_user->CheckExchange(EXCHANGE_ABYSS_GEM));
93
94 // EXCHANGE_WARRIOR_BEGINNER grants the following items:
95 // 120110992 Beginner's Weapon
96 // 389011000 Water of life (x10)
97 // 389017000 Potion of intelligence (x10)
98 // If we free up 3 empty slots, then add the potions such that there
99 // is only one slot, the check should pass.
100 _user->TestRobItem(ITEM_GOLD_BAR, 1);
101 _user->TestRobItem(ITEM_GOLD_BAR, 1);
102 _user->TestRobItem(ITEM_GOLD_BAR, 1);
103 _user->TestGiveItem(ITEM_HP_POTION, 5);
104 _user->TestGiveItem(ITEM_MP_POTION, 5);
105 ASSERT_EQ(_user->GetNumberOfEmptySlots(), 1);
106 EXPECT_TRUE(_user->CheckExchange(EXCHANGE_WARRIOR_BEGINNER));
107}

109// This test covers CUser::RunExchange for a EXCHANGE_TYPE_ALL_ITEMS type.
110// RunExchange checks for inputs, robs them, then gives the appropriate
111// exchange reward. This function does not return a status, so testing
112// is done via inventory inspection.
113TEST_F(ItemExchangeTest, RunExchange_EXCHANGE_TYPE_ALL_ITEMS)
114{
115 _user->ClearInventory();
116 _user->ResetSend();
117
118 // load exchange data
119 auto* exchange = _app->m_ItemExchangeMap.GetData(EXCHANGE_WARRIOR_BEGINNER);
120 ASSERT_TRUE(exchange != nullptr);
121
122 ItemPair originItems[MAX_EXCHANGE_ITEMS] {};
123 ItemPair exchangeItems[MAX_EXCHANGE_ITEMS] {};
124 for (int i = 0; i < MAX_EXCHANGE_ITEMS; i++)
125 {
126 originItems[i].ItemId = exchange->OriginItemNumber[i];
127 originItems[i].Count = exchange->OriginItemCount[i];
128 exchangeItems[i].ItemId = exchange->ExchangeItemNumber[i];
129 exchangeItems[i].Count = exchange->ExchangeItemCount[i];
130 }
131
132 // Give the player required origin items for the beginner quest
133 for (ItemPair item : originItems)
134 {
135 _user->TestGiveItem(item.ItemId, item.Count);
136 }
137
138 // RUN_EXCHANGE - EXCHANGE_TYPE_ALL_ITEMS
139 // will check and rob origin items then give all the exchange items
140 // stub RobItem callbacks
141 for (ItemPair item : originItems)
142 {
143 if (item.ItemId <= 0 || item.Count <= 0)
144 continue;
145
146 _user->AddSendCallback(
147 [=](const char* pBuf, int len)
148 {
149 ASSERT_EQ(len, sizeof(WeightChangePacket));
150 auto packet = reinterpret_cast<const WeightChangePacket*>(pBuf);
151 EXPECT_EQ(packet->Opcode, WIZ_WEIGHT_CHANGE);
152 });
153 _user->AddSendCallback(
154 [=](const char* pBuf, int len)
155 {
156 ASSERT_EQ(len, sizeof(ItemCountChangePacket));
157 auto packet = reinterpret_cast<const ItemCountChangePacket*>(pBuf);
158 EXPECT_EQ(packet->Opcode, WIZ_ITEM_COUNT_CHANGE);
159 EXPECT_EQ(packet->ItemId, item.ItemId);
160 });
161 }
162
163 // stub GiveItem callbacks
164 for (ItemPair item : exchangeItems)
165 {
166 if (item.ItemId <= 0 || item.Count <= 0)
167 continue;
168
169 _user->AddSendCallback(
170 [=](const char* pBuf, int len)
171 {
172 ASSERT_EQ(len, sizeof(WeightChangePacket));
173 auto packet = reinterpret_cast<const WeightChangePacket*>(pBuf);
174 EXPECT_EQ(packet->Opcode, WIZ_WEIGHT_CHANGE);
175 });
176 _user->AddSendCallback(
177 [=](const char* pBuf, int len)
178 {
179 ASSERT_EQ(len, sizeof(ItemCountChangePacket));
180 auto packet = reinterpret_cast<const ItemCountChangePacket*>(pBuf);
181 EXPECT_EQ(packet->Opcode, WIZ_ITEM_COUNT_CHANGE);
182 EXPECT_EQ(packet->ItemId, item.ItemId);
183 EXPECT_EQ(packet->ItemCount, item.Count);
184 });
185 }
186
187 _user->RunExchange(EXCHANGE_WARRIOR_BEGINNER);
188
189 EXPECT_TRUE(_user->CheckExistItemAnd(exchangeItems));
190}
